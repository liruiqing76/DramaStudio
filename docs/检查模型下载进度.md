在Windows上运行以下命令（需要先配置SSH密钥或知道密码）：

```powershell
# 每30秒检查一次
while($true) {
    ssh root@116.172.93.163 "ls -lh /root/comfyui/models/checkpoints/ | grep -i ltx"
    Write-Host "--- $(Get-Date) ---"
    Start-Sleep -Seconds 30
}
```
